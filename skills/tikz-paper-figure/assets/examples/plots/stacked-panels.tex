% stacked-panels: two quantities measured over the same x (accuracy and cost against model size), drawn as two
% panels stacked on one shared x axis. This is the replacement for a chart with two y axes: a second y axis has
% no natural alignment with the first, so any crossing it shows is an artefact of the scales chosen. The x
% descriptions are set once, at the bottom; each panel has its own y range and a horizontal y label; the same
% system keeps its colour and mark in both. (Wilke, Visualizing trends; Coordinate systems and axes)
\documentclass[10pt,tikz,border=2pt]{standalone}
\usepackage{plotfig}
\begin{document}
\begin{tikzpicture}
% the vertical sep holds the lower panel's horizontal y label (7pt plus 3pt) under the upper panel's tick marks
\begin{groupplot}[group style={group size=1 by 2, vertical sep=0.65cm, x descriptions at=edge bottom},
  paper, width=6cm, y label top,
  xmode=log, xtick={0.5,1,3,7,14,30}, xticklabels={0.5,1,3,7,14,30}, xlabel={parameters (B)},
  % explicit limits: a little air left of 0.5 and room for the direct labels after 30 (one `enlarge x limits`
  % call cannot set two different amounts, and a second call replaces the first)
  xmin=0.42, xmax=90]
% ------------------------------------------------------------------ top: the quality
% heights differ per panel (the quantity the paragraph is about gets more room); the shared x keeps them aligned
\nextgroupplot[height=2.6cm, ylabel={accuracy (\%)}, ymin=35, ymax=90, ytick={40,60,80}]
\addplot[kept, mark=*, mark size=1.3pt] coordinates {(0.5,45) (1,54) (3,65) (7,74) (14,79) (30,83)} node[dl, text=kept, yshift=1.5pt] {Ours};
