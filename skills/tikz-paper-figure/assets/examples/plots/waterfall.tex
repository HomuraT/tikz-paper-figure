% waterfall: how the components add up from the baseline to our final score. The first and last bars are
% amounts and start at zero; the floating bars between them are the deltas of adding one component at a time,
% in the order they were added (an ordered variable, never sorted), blue when positive and orange when negative,
% thin connectors carrying the running total from row to row. The ablation chart (ablation.tex) is the same
% question asked backwards (remove one component from the full model). Catalogues call this a bridge chart.
\documentclass[10pt,tikz,border=2pt]{standalone}
\usepackage{plotfig}
\begin{document}
\begin{tikzpicture}
\begin{axis}[paper, width=5.8cm, height=2.7cm,

  xmajorgrids, ymajorgrids=false, y axis hidden, axis x line*=bottom,
  ymin=0.4, ymax=6.6, ytick={1,...,6},
  yticklabels={Ours, + tool calls, + self-check, + reranking, + retrieval, Baseline},
  y tick label style={text=ink, font=\fontsize{7}{8}\selectfont}]
\addplot[draw=none, forget plot] coordinates {(0,1) (80,6)};
% full bars: amounts from zero
\fill[black!30] (axis cs:0,5.83) rectangle (axis cs:61.2,6.17);
\fill[barblue]  (axis cs:0,0.83) rectangle (axis cs:73.5,1.17);
% floating bars: the deltas, drawn from the previous running total
\fill[barblue!65] (axis cs:61.2,4.83) rectangle (axis cs:68.6,5.17);

\end{axis}
