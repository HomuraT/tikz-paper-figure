% Candidate 8: composition donuts, one per dimension of a dataset, the total in the hole, labels outside with
% percentages. Slices in the palette order, white gaps between slices. Pure TikZ, no pgfplots.
\documentclass[10pt,tikz,border=2pt]{standalone}
\usepackage{plotfig}
% \donut{position}{centre text}{title}{value/colour/label, ...}   values in percent, drawn clockwise from the top
\newcommand{\donut}[4]{%
  \begin{scope}[shift={(#1)}]
    \xdef\dA{90}
    \foreach \v/\c/\l in {#4} {
      \pgfmathsetmacro{\dB}{\dA-\v*3.6}
      \fill[\c, draw=white, line width=1pt] (\dA:0.62) arc (\dA:\dB:0.62) -- (\dB:1.0) arc (\dB:\dA:1.0) -- cycle;
      \pgfmathsetmacro{\dM}{(\dA+\dB)/2}
      \pgfmathsetmacro{\dX}{cos(\dM)}
      \ifdim\dX pt<0pt \def\anc{east}\else \def\anc{west}\fi
      \node[font=\fontsize{6.5}{7}\selectfont, text=ink, anchor=\anc, inner sep=1pt] at (\dM:1.12) {\l\ {\color{black!55}\v\%}};
      \xdef\dA{\dB}
    }
    \node[font=\fontsize{6.5}{7}\selectfont, text=ink, align=center] at (0,0) {#2};
    \node[font=\fontsize{8}{9}\selectfont\bfseries, text=ink, anchor=south] at (0,1.25) {#3};
  \end{scope}}
