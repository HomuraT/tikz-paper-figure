% Example 2: a method figure whose cards are the stages of a procedure, read left to right through the
% labelled stage arrows in the gaps. Textbook data: two tables about employees and their units.
% Card 1: the source tables, one of them not used (grey). Card 2: the entities extracted from them, with the
% key of each entity marked and dotted arrows back to the tables. Card 3: what the last stage adds (orange, dashed).
% The footer bands read as one sentence across the three cards. The legend replaces a colour key in the caption.
% Build:  python <skill>/scripts/build_figure.py pipeline-stages.tex
\documentclass[10pt,tikz,border={1pt 3.5pt 3pt 1.5pt}]{standalone}
\usepackage{cardfig}
\setlength{\cardw}{3.95cm}
\setlength{\cardh}{4.55cm}
\setlength{\cardgap}{0.95cm}
\begin{document}
\begin{tikzpicture}
\cardpanels{3}
\cardhead{P1}{Source tables}  \cardicon{P1}{database}
\cardhead{P2}{Entities}       \cardicon{P2}{cubes}
\cardhead{P3}{Model}          \cardicon{P3}{sitemap}
\cardfoot{P1}{all tables of the source}
\cardfoot{P2}{those that hold an entity}
\cardfoot{P3}{plus the classes the merge adds}
\cardstage{P1}{P2}{extract}
\cardstage{P2}{P3}{merge}
% ================= card 1: the tables as found; the log table is not used and stays grey
\dbicon[0.72]{$(P1.north west)+(0.25cm,-1.2cm)$}{dbA}
\node[tname, text=dbA, anchor=south west] at ($(P1.north west)+(0.53cm,-1.16cm)$) {emp};
\matrix[tblA, anchor=north west, column 1/.style={nodes={text width=0.3cm}}, column 2/.style={nodes={text width=0.55cm}}, column 3/.style={nodes={text width=0.45cm}}]
  at ($(P1.north west)+(0.25cm,-1.24cm)$) (aemp) {
  id & name & unit \\
  7 & Ada & 2 \\
  9 & Bob & 2 \\
};
\boxshadow[]{aemp}
\node[tname, text=dbA, anchor=south east] at ($(P1.north east)+(-0.25cm,-1.16cm)$) (aun) {unit};
\dbicon[0.72]{$(aun.south west)+(-0.28cm,-0.04cm)$}{dbA}
\matrix[tblA, anchor=north east, column 1/.style={nodes={text width=0.3cm}}, column 2/.style={nodes={text width=0.55cm}}]
  at ($(P1.north east)+(-0.25cm,-1.24cm)$) (aunit) {
  id & name \\
  2 & R\&D \\
  3 & Ops \\
};
\boxshadow[]{aunit}
\draw[fk] (aemp-1-3.east) -- (aunit-1-1.west);
\node[tname, text=black!50, anchor=south west] at ($(P1.north west)+(0.25cm,-2.76cm)$) {log};
\matrix[tbl, anchor=north west, nodes={cell, faint, fill=black!3}, column 1/.style={nodes={text width=0.55cm}}, column 2/.style={nodes={text width=1.25cm}}]
  at ($(P1.north west)+(0.25cm,-2.84cm)$) (alog) {
  ts & message \\
  09:12 & login ok \\
};
\boxshadow[]{alog}
% ================= card 2: the entities above, the tables they come from below, dotted arrows between them
\classbox[cls, anchor=north west]{bemp}{$(P2.north west)+(0.25cm,-0.95cm)$}{Employee}{\keyic name\\ \nokey unit}
\classbox[cls, anchor=north east]{bun}{$(P2.north east)+(-0.25cm,-0.95cm)$}{Unit}{\keyic name}
\draw[obj] (bemp.east |- bun.text east) -- node[lab, above] {inUnit} (bun.west |- bun.text east);
\dbicon[0.72]{$(P2.north west)+(0.25cm,-2.66cm)$}{dbA}
\node[tname, text=dbA, anchor=south west] at ($(P2.north west)+(0.53cm,-2.62cm)$) (ben) {emp};
\matrix[tblA, anchor=north west, column 1/.style={nodes={text width=0.3cm}}, column 2/.style={nodes={text width=0.55cm}}, column 3/.style={nodes={text width=0.45cm}}]
  at ($(P2.north west)+(0.25cm,-2.7cm)$) (bemp2) {
  id & name & unit \\
  7 & Ada & 2 \\
  9 & Bob & 2 \\
};
\boxshadow[]{bemp2}
\node[tname, text=dbA, anchor=south east] at ($(P2.north east)+(-0.25cm,-2.62cm)$) (bun2n) {unit};
\dbicon[0.72]{$(bun2n.south west)+(-0.28cm,-0.04cm)$}{dbA}
\matrix[tblA, anchor=north east, column 1/.style={nodes={text width=0.3cm}}, column 2/.style={nodes={text width=0.55cm}}]
  at ($(P2.north east)+(-0.25cm,-2.7cm)$) (bunit) {
  id & name \\
  2 & R\&D \\
  3 & Ops \\
};
\boxshadow[]{bunit}
\draw[fk] (bemp2-1-3.east) -- (bunit-1-1.west);
\draw[pop] (ben.north) to[bend left=12] (bemp.south);
\draw[pop] (bun2n.north) to[bend right=12] (bun.south);
\coordinate (fkmid) at ($(bemp2-1-3.east)!0.5!(bunit-1-1.west)$);
\coordinate (inmid) at ($(bemp.east |- bun.text east)!0.5!(bun.west |- bun.text east)$);
\draw[pop] (fkmid) to[bend right=12] (inmid);
% ================= card 3: the merge adds a superclass and the subclass axiom (orange, dashed)
\classtag[addedcls1, anchor=north]{cper}{$(P3.north west)+(1.1cm,-1.35cm)$}{Person}
\classbox[cls, anchor=north west]{cemp}{$(P3.north west)+(0.25cm,-2.3cm)$}{Employee}{\keyic name\\ \nokey unit}
\classbox[cls, anchor=north east]{cun}{$(P3.north east)+(-0.25cm,-2.3cm)$}{Unit}{\keyic name}
\draw[sub, added] (cemp.north -| cper.south) -- (cper.south);
\draw[obj] (cemp.east |- cun.text east) -- node[lab, above] {inUnit} (cun.west |- cun.text east);
% ================= legend
\cardlegend{P2}{%
  \legswatch{draw=kept, fill=keptfill} extracted entity\legsep
  \legswatch{faint, fill=black!7} not used\legsep
  \legswatch{added, fill=addfill} added by the merge\legsep
  \legline{pop} derived from\legsep
  {\color{kept}\faIcon{key}}\, key}
\end{tikzpicture}
\end{document}
