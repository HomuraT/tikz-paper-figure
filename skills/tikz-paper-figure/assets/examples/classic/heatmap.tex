% heatmap: yield over a temperature x catalyst-loading screen, in the classic (Matplotlib) style, polished after
% the gallery's "Creating annotated heatmaps": imshow with viridis (vmin=0, vmax=100), white lines between the
% cells (a minor grid at the cell edges, drawn on top), no spines and no ticks, the value in every cell in white
% on the dark half and black on the light half, and a colour bar with its label. The best cell (91 % at 70 C,
% 5 mol %) is outlined and set in bold; above 70 C the product degrades, which the colours show as a ridge.
% Values from data/make_data.py.
\documentclass[10pt,tikz,border=4pt]{standalone}
\usepackage{classicfig}
\begin{document}
\begin{tikzpicture}
\begin{axis}[classic, mpl imshow, width=5.4cm, height=4.7cm,
  title={Yield screen},
  xlabel={Catalyst loading (mol\%)}, ylabel={Temperature (\textdegree C)},
  xmin=-0.5, xmax=5.5, ymin=-0.5, ymax=5.5,
  xtick={0,...,5}, xticklabels={0.5,1,2,3,4,5}, ytick={0,...,5}, yticklabels={40,50,60,70,80,90},
  % no spines, no tick marks; white lines between the cells
  axis line style={draw=none}, major tick length=0pt,
  minor xtick={-0.5,0.5,...,5.5}, minor ytick={-0.5,0.5,...,5.5}, minor tick length=0pt,
  xminorgrids, yminorgrids, minor grid style={white, line width=2pt},
  colormap name=viridis, point meta min=0, point meta max=100,
  colorbar, classic colorbar={Yield (\%)}, colorbar style={ytick={0,20,...,100}, axis line style={draw=none}},
  % white text on the dark half, bold for the best cell (the value is saved first: \color overwrites
  % \pgfmathresult; digits in text mode, since math mode ignores \bfseries)
  nodes near coords={\pgfmathfloattofixed{\pgfplotspointmeta}\let\cellvalue\pgfmathresult
    \ifdim\cellvalue pt<60pt\color{white}\else\color{black}\fi
    \ifdim\cellvalue pt>90pt\bfseries\fi
    \pgfmathprintnumber[fixed, precision=0, assume math mode=true]{\pgfplotspointmeta}},
  every node near coord/.append style={font=\fontsize{9}{11}\selectfont, anchor=center}]
\addplot[matrix plot*, mesh/cols=6, point meta=explicit] coordinates {
  (0,0) [10] (1,0) [19] (2,0) [35] (3,0) [47] (4,0) [57] (5,0) [65]

  (0,1) [14] (1,1) [27] (2,1) [46] (3,1) [60] (4,1) [70] (5,1) [77]

  (0,2) [19] (1,2) [34] (2,2) [56] (3,2) [70] (4,2) [80] (5,2) [86]

  (0,3) [23] (1,3) [41] (2,3) [65] (3,3) [78] (4,3) [86] (5,3) [91]

  (0,4) [25] (1,4) [42] (2,4) [63] (3,4) [74] (4,4) [80] (5,4) [83]

  (0,5) [20] (1,5) [33] (2,5) [47] (3,5) [53] (4,5) [56] (5,5) [57]
};
% the best cell: outlined inside its white border (the grid is drawn on top of everything), value in bold
\draw[black, line width=1.6pt] (axis cs:4.57,2.57) rectangle (axis cs:5.43,3.43);
\end{axis}
\end{tikzpicture}
\end{document}
